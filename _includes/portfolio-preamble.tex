% AI-owned: LaTeX tweaks for the PDF build of teaching-portfolio.qmd.

\usepackage{needspace}
\usepackage{etoolbox}
\usepackage{titlesec}

% Headings map as `##` \subsection (Courses, Mentoring, ...), `###`
% \subsubsection (an institution), `####` \paragraph (one course), `#####`
% \subparagraph (a course's notes). Both paragraph levels are run-in
% headings by default; make them display headings so a table or a bullet
% list starts on the line below.
\titleformat{\subsubsection}{\normalfont\large\bfseries}{}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{3.5ex plus 1ex}{1.2ex}
\titleformat{\paragraph}{\normalfont\normalsize\bfseries}{}{0pt}{}
\titlespacing*{\paragraph}{0pt}{2.2ex plus .8ex}{0.8ex}
\titleformat{\subparagraph}{\normalfont\normalsize\itshape}{}{0pt}{}
\titlespacing*{\subparagraph}{0pt}{1.2ex plus .5ex}{0.4ex}

% Keep a course heading on the same page as its table (a table never breaks
% before its first rows anyway), and a notes heading with its first lines.
\pretocmd{\subsection}{\needspace{9\baselineskip}}{}{}
\pretocmd{\subsubsection}{\needspace{9\baselineskip}}{}{}
\pretocmd{\paragraph}{\needspace{9\baselineskip}}{}{}
\pretocmd{\subparagraph}{\needspace{4\baselineskip}}{}{}

% Tables flush left with the prose instead of centred at varying widths.
\setlength{\LTleft}{0pt}
\setlength{\LTright}{\fill}
